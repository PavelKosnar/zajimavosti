#include <cassert>
#include <vector>
#include <algorithm>

using namespace std;

/*
 * CVICENI 7 - DYNAMICKE PROGRAMOVANI
 * Optimalizace obsahu cache
 *
 * ZADANI:
 *
 * Aplikace pracuje s objekty ulozenymi na pomalejsim ulozisti.
 * Vybrane objekty lze uchovavat v rychle cache, ktera ma
 * omezenou kapacitu.
 *
 * Kazdy objekt ma dve vlastnosti:
 *    size    - velikost objektu v MB
 *    benefit - prinos ulozeni objektu do cache
 *
 * Ukolem je vybrat objekty tak, aby jejich celkovy prinos
 * byl maximalni a soucet jejich velikosti neprekrocil
 * kapacitu cache.
 *
 * Kazdy objekt lze vybrat nejvyse jednou.
 * Objekty nelze delit na mensi casti.
 *
 * IMPLEMENTACE:
 *
 * 1. Navrhnete algoritmus pro nalezeni maximalniho prinosu.
 * 2. Implementujte funkci maximizeBenefit().
 * 3. Overte spravnost pomoci pripravenych testu.
 * 4. Odpovezte na otazky na konci souboru.
 *
 * Predpokladejte nezaporne hodnoty size, benefit a capacity.
 * Funkce vraci maximalni dosazitelny prinos.
 * Pokud nelze vybrat zadny objekt, vraci 0.
 */

struct Object {
    int size;
    int benefit;
};

// ---------------------------------------------------------
// UKOL: Implementujte nasledujici funkci.
// ---------------------------------------------------------

int maximizeBenefit(const vector<Object>& objects, int capacity) {
    if (capacity <= 0 || objects.empty()) {
        return 0;
    }

    // dp[w] uchovava maximalni prinos pro kapacitu w
    vector<int> dp(capacity + 1, 0);

    for (const auto& obj : objects) {
        // Prochazime kapacitu zprava doprava, aby se kazdy objekt
        // pouzil v dane iteraci nejvyse jednou (0/1 batoh).
        for (int w = capacity; w >= obj.size; --w) {
            dp[w] = max(dp[w], dp[w - obj.size] + obj.benefit);
        }
    }

    return dp[capacity];
}

// ---------------------------------------------------------
// TESTY - neupravujte ocekavane vysledky.
// ---------------------------------------------------------

int main() {

    // Test 1: Zakladni priklad
    {
        vector<Object> objects = {
            {6, 12},
            {4, 9},
            {5, 11},
            {3, 5}
        };

        assert(maximizeBenefit(objects, 10) == 21);
    }

    // Test 2: Hladovy vyber podle pomeru
    // prinos/velikost nemusi byt optimalni.
    {
        vector<Object> objects = {
            {6, 12},
            {5, 9},
            {5, 9}
        };

        assert(maximizeBenefit(objects, 10) == 18);
    }

    // Test 3: Vsechny objekty se vejdou
    {
        vector<Object> objects = {
            {2, 4},
            {3, 5},
            {4, 7}
        };

        assert(maximizeBenefit(objects, 10) == 16);
    }

    // Test 4: Zadny objekt se nevejde
    {
        vector<Object> objects = {
            {6, 10},
            {8, 15}
        };

        assert(maximizeBenefit(objects, 3) == 0);
    }

    // Test 5: Nulova kapacita
    {
        vector<Object> objects = {
            {2, 5},
            {3, 8}
        };

        assert(maximizeBenefit(objects, 0) == 0);
    }

    // Test 6: Prazdny seznam objektu
    {
        vector<Object> objects;

        assert(maximizeBenefit(objects, 10) == 0);
    }

    // Test 7: Kazdy objekt lze vybrat pouze jednou
    {
        vector<Object> objects = {
            {3, 10}
        };

        assert(maximizeBenefit(objects, 9) == 10);
    }

    // Test 8: Vyber vice mensich objektu
    {
        vector<Object> objects = {
            {8, 15},
            {4, 9},
            {3, 8},
            {3, 8}
        };

        assert(maximizeBenefit(objects, 10) == 25);
    }

    // Test 9: Optimalni reseni nemusi
    // vyuzit celou kapacitu cache.
    {
        vector<Object> objects = {
            {6, 15},
            {4, 8},
            {3, 7}
        };

        assert(maximizeBenefit(objects, 8) == 15);
    }

    return 0;
}

/*
 * =========================================================
 * OTAZKY
 * =========================================================
 *
 * Odpovedi zapiste primo pod jednotlive otazky.
 *
 * 1. PRINCIP ALGORITMU
 *
 * Vysvetlete, jak vas algoritmus postupuje.
 * Jake mensi podproblemy resi a jak z jejich vysledku
 * sestavuje optimalni reseni?
 *
 * Odpoved:
 * Algoritmus vyuziva dynamicke programovani pro reseni tzv. 0/1 problemu batohu.
 * Podproblem spocite v urceni maximalniho dosazitelneho prinosu pro danou
 * dilci kapacitu w (od 0 do capacity) pri zvazovani prvnich i objektu.
 *
 * Pro kazdy objekt se rozhodujeme mezi dvěma moznostmi:
 * 1. Objekt do cache NEVLOZIME: prinos pro kapacitu w zustava roven
 *    dosavadnimu prinosu bez tohoto objektu.
 * 2. Objekt do cache VLOZIME (pokud obj.size <= w): prinos bude roven souctu
 *    prinosu aktualniho objektu a optimalniho prinosu pro zbivajici kapacitu
 *    (w - obj.size) dosazeneho v predchozim kroku.
 *
 * Z techto dvoii moznosti zvolime tu, ktera dava vyssi prinos:
 * dp[w] = max(dp[w], dp[w - obj.size] + obj.benefit).
 *
 * 2. POMOCNE DATOVE STRUKTURY
 *
 * Jake pomocne datove struktury pouzivate?
 * Vysvetlete, co predstavuji hodnoty, ktere do nich
 * ukladate.
 *
 * Odpoved:
 * Pouzivame jednorozmerne pole (std::vector<int> dp) o velikosti (capacity + 1).
 *
 * - Index `w` v poli reprezentuje konkretni kapacitu cache (v rozmedzi 0 az capacity).
 * - Hodnota `dp[w]` uchovava maximalni prinos, ktereho lze dosahnout s vyuzitim
 *   dosud zpracovanych objektu pro kapacitu `w`.
 *
 * Diky prohledavani kapacity vnitrniho cyklu v sestupnem poradi (od capacity dolu
 * k obj.size) si vystacime s 1D polem misto 2D matice, protoze neprepijeme hodnoty
 * z predchozi iterace dřív, nez je pouzijeme.
 *
 * 3. CASOVA SLOZITOST
 *
 * Urcete casovou slozitost vaseho algoritmu.
 * Oznacte:
 *    N - pocet objektu
 *    C - kapacita cache
 *
 * Vyjadrete slozitost pomoci O(...) a vysvetlete,
 * z ceho vyplyva.
 *
 * Odpoved:
 * Casova slozitost je O(N * C).
 *
 * Algoritmus obsahuje dva vnorene cykly:
 * - Vnejsi cyklus se provede N-krat (pro kazdy objekt ze seznamu).
 * - Vnitrni cyklus se pro kazdy objekt provede maximalne (C + 1)-krat.
 * Uvnitr vnitrniho cyklu se provadeji pouze konstantni operace (porovnani,
 * odcitani, prirazeni) v case O(1). Celkovy pocet operaci je tedy umerny N * C.
 *
 * 4. PAMETOVA SLOZITOST
 *
 * Urcete pametovou slozitost vaseho algoritmu
 * vzhledem k N a C.
 *
 * Vysvetlete, na cem zavisi mnozstvi pouzite pameti.
 *
 * Odpoved:
 * Pametova slozitost je O(C).
 *
 * Mnozstvi pouzite pameti zavisi pouze na kapacite cache C, protoze pouzivame
 * jednorozmerny vector `dp` o velikosti (C + 1). Pametova slozitost je nezavisla
 * na poctu objektu N diky prostorove optimalizaci z 2D matice na 1D pole.
 *
 * 5. HLADOVY ALGORITMUS
 *
 * Predstavte si, ze objekty seradite podle pomeru
 * benefit / size a budete je postupne vybirat,
 * dokud se vejdou do cache.
 *
 * Zaruci tento postup vzdy optimalni reseni?
 * Vysvetlete svou odpoved a uvedte protipriklad.
 *
 * Odpoved:
 * NE, tento postup NEZARUCI vzdy optimalni reseni.
 * Hladovy postup funguje pouze pro spojity (fractional) problem batohu,
 * kde lze objekty delit. U 0/1 verze muze hladovy vyber zabrat kapacitu
 * objektu s mírně vyhodnejsim pomerem, ale znemoznit kombinaci jinych objektu,
 * ktere by dohromady vyuzily kapacitu lépe a s celkově vyšším přínosem.
 *
 * Protipriklad (odpovidajici Testu 2):
 * Kapacita C = 10.
 * Objekty:
 *   - Objekt A: size = 6, benefit = 12 (pomer benefit/size = 2.0)
 *   - Objekt B: size = 5, benefit = 9  (pomer benefit/size = 1.8)
 *   - Objekt C: size = 5, benefit = 9  (pomer benefit/size = 1.8)
 *
 * Hladovy postup podle pomeru vybere nejprve Objekt A (pomer 2.0).
 * Zbyvajici kapacita je 10 - 6 = 4. Zadne dalsi objekty se uz nevejdou.
 * Celkovy prinos hladoveho reseni = 12.
 *
 * Optimalni reseni vybere Objekty B a C (celkova velikost 5 + 5 = 10 <= 10).
 * Celkovy prinos optimalniho reseni = 9 + 9 = 18.
 */
